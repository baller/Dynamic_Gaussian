# W-CVCT-GS Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add wavelet-supervised sub-Gaussian disentanglement (FDSG) and cross-view color triangulation (CVCT) on top of the existing StereoGS pipeline (`train_stereo_gs.py` / `lib/stereo_gs/`), with no changes to architecture — only new losses, a new color head, and a three-phase training schedule.

**Architecture:** Two innovations layered on the StereoGS pipeline:
- **FDSG** binds each `k_sub` sub-Gaussian level to one Haar-DWT detail band via a render-attribution loss (`L_disentangle`), plus a multi-band reconstruction loss (`L_band`) and a GT-wavelet-guided sparsity (`L_active`) that replaces the existing CAGS sparsity.
- **CVCT** computes per-Gaussian color as `ω·c_left + (1-ω)·c_right_warped + Δrgb` with a learned visibility gate `ω` and bounded residual `Δrgb`, supervised by a cross-view photometric consistency loss (`L_cycle`) and visibility soft alignment (`L_omega_align`).

Both innovations close into a single supervisory loop through the wavelet decomposition.

**Tech Stack:** PyTorch 2.x, existing `diff_gaussian_rasterization` backend, Haar DWT implemented as plain tensor ops (no `pytorch_wavelets` dependency), pytest for unit tests.

**Spec reference:** `docs/superpowers/specs/2026-04-25-w-cvct-gs-design.md`

```python
"""Haar discrete wavelet transform utilities for W-CVCT-GS.

Implements an orthonormal Haar 2D DWT as plain tensor ops so the project
keeps zero new external dependencies. Used by the FDSG losses to decompose
images into LL + 3 detail bands per level.
"""

from __future__ import annotations

from typing import Tuple, Dict

import torch
import torch.nn.functional as F


def haar_dwt2d_step(x: torch.Tensor):
    """One level of orthonormal Haar 2D DWT.

    Args:
        x: (B, C, H, W) with H, W even.

    Returns:
        (LL, LH, HL, HH), each (B, C, H/2, W/2).
    """
    if x.shape[-1] % 2 != 0 or x.shape[-2] % 2 != 0:
        raise ValueError(f"haar_dwt2d_step needs even H,W, got {tuple(x.shape)}")
    a = x[..., 0::2, 0::2]
    b = x[..., 0::2, 1::2]
    c = x[..., 1::2, 0::2]
    d = x[..., 1::2, 1::2]
    LL = (a + b + c + d) * 0.5
    LH = (a + b - c - d) * 0.5
    HL = (a - b + c - d) * 0.5
    HH = (a - b - c + d) * 0.5
    return LL, LH, HL, HH


def dwt3(x: torch.Tensor) -> Dict[str, object]:
    """3-level Haar DWT cascade.

    Args:
        x: (B, C, H, W) with H, W divisible by 8.

    Returns:
        dict with:
            'LL':  (B, C, H/8, W/8)
            'D1':  (LH_1, HL_1, HH_1) at H/2, W/2
            'D2':  (LH_2, HL_2, HH_2) at H/4, W/4
            'D3':  (LH_3, HL_3, HH_3) at H/8, W/8
    """
    LL_1, LH_1, HL_1, HH_1 = haar_dwt2d_step(x)
    LL_2, LH_2, HL_2, HH_2 = haar_dwt2d_step(LL_1)
    LL_3, LH_3, HL_3, HH_3 = haar_dwt2d_step(LL_2)
    return {
        "LL": LL_3,
        "D1": (LH_1, HL_1, HH_1),
        "D2": (LH_2, HL_2, HH_2),
        "D3": (LH_3, HL_3, HH_3),
    }


def pad_to_multiple(x: torch.Tensor, m: int) -> Tuple[torch.Tensor, Tuple[int, int, int, int]]:
    """Reflect-pad spatial dims so H, W are multiples of m.

    Args:
        x: (B, C, H, W).
        m: alignment factor (8 for 3-level DWT).

    Returns:
        (padded, (pad_left, pad_right, pad_top, pad_bottom))
    """
    _, _, H, W = x.shape
    pad_h = (m - H % m) % m
    pad_w = (m - W % m) % m
    pad = (0, pad_w, 0, pad_h)
    if pad_h == 0 and pad_w == 0:
        return x, pad
    return F.pad(x, pad, mode="reflect"), pad


def band_energy(detail_band: Tuple[torch.Tensor, torch.Tensor, torch.Tensor]) -> torch.Tensor:
    """Per-pixel L2 magnitude of a (LH, HL, HH) triple, averaged over channels.

    Args:
        detail_band: tuple of three tensors (B, C, h, w).

    Returns:
        (B, 1, h, w) energy map.
    """
    LH, HL, HH = detail_band
    sq = LH.pow(2) + HL.pow(2) + HH.pow(2)
    e = torch.sqrt(sq + 1e-12).mean(dim=1, keepdim=True)
    return e


def log_compress(x: torch.Tensor, k: float = 10.0) -> torch.Tensor:
    """Sign-preserving log compression: y = sign(x) * log1p(|x|*k).

    Used to stabilize losses on small-magnitude high-frequency bands.
    """
    return torch.sign(x) * torch.log1p(x.abs() * k)
```

```python
"""Frequency-domain losses for FDSG (W-CVCT-GS).

Three losses:
  - l_band:        multi-band reconstruction (L_band)
  - l_active:      GT-wavelet-guided sub-Gaussian sparsity (L_active)
  - l_disentangle: per-level sub-Gaussian band specialization (L_disentangle)
"""

from __future__ import annotations

from typing import Sequence, Tuple

import torch
import torch.nn.functional as F

from lib.stereo_gs.wavelet_ops import (
    dwt3,
    pad_to_multiple,
    band_energy,
    log_compress,
)


def _dwt3_padded(img: torch.Tensor):
    """Pad to multiple of 8 then DWT3. Returns (dwt_dict, pad)."""
    padded, pad = pad_to_multiple(img, 8)
    return dwt3(padded), pad


def l_band(
    pred: torch.Tensor,
    gt: torch.Tensor,
    ll_weight: float = 0.4,
    band_weights: Sequence[float] = (0.3, 0.2, 0.1),
    log_compress_k: float = 10.0,
) -> torch.Tensor:
    """Multi-band L1 reconstruction loss.

    L_band = β·‖LL−LL*‖₁ + Σ_j α_j · ‖D_j − D_j*‖₁ (in log-compressed space).

    Args:
        pred, gt: (B, 3, H, W) in any consistent range; spec recommends [0, 1].
        ll_weight: β.
        band_weights: (α_1, α_2, α_3).
        log_compress_k: log-compression strength.

    Returns:
        scalar tensor.
    """
    assert pred.shape == gt.shape
    assert len(band_weights) == 3
    pred_dwt, _ = _dwt3_padded(pred)
    gt_dwt, _ = _dwt3_padded(gt)

    loss = ll_weight * F.l1_loss(pred_dwt["LL"], gt_dwt["LL"])
    for j, w in enumerate(band_weights, start=1):
        if w == 0.0:
            continue
        for sub_pred, sub_gt in zip(pred_dwt[f"D{j}"], gt_dwt[f"D{j}"]):
            sp = log_compress(sub_pred, k=log_compress_k)
            sg = log_compress(sub_gt, k=log_compress_k)
            loss = loss + w * F.l1_loss(sp, sg)
    return loss
```

```python
def l_active(
    split_weights: torch.Tensor,
    gt_image: torch.Tensor,
    gt_levels: int = 3,
) -> torch.Tensor:
    """GT-wavelet-guided sparsity loss for sub-Gaussian split weights.

    For each level j, sub-Gaussian j may activate only where GT has D_j energy.
    L_active = Σ_j ‖ weights[:, j-1] · (1 − E_j_GT) ‖_1.

    Args:
        split_weights: (B, k_sub, H, W) ∈ [0, 1] — output of LearnedSplitCriterion.
            k_sub must equal `gt_levels`.
        gt_image: (B, 3, H, W).
        gt_levels: number of wavelet levels to use (default 3).

    Returns:
        scalar tensor.
    """
    B, k_sub, H, W = split_weights.shape
    assert k_sub == gt_levels, f"k_sub={k_sub} must match gt_levels={gt_levels}"

    gt_dwt, _ = _dwt3_padded(gt_image)
    loss = split_weights.new_zeros(())
    for j in range(1, gt_levels + 1):
        e_j = band_energy(gt_dwt[f"D{j}"])  # (B, 1, h_j, w_j)
        e_j_full = F.interpolate(e_j, size=(H, W), mode="bilinear", align_corners=False)
        e_max = e_j_full.amax(dim=(2, 3), keepdim=True).clamp(min=1e-6)
        e_norm = (e_j_full / e_max).clamp(0, 1)
        w_j = split_weights[:, j - 1 : j]
        loss = loss + (w_j * (1.0 - e_norm)).abs().mean()
    return loss
```

```python
def pts2render_cags_per_level(data, bg_color, level: str = "all"):
    """Render a CAGS scene including only Gaussians at the specified level(s).

    Used by L_disentangle: rendering with each sub-Gaussian level removed yields
    that level's contribution image ΔI_j.

    Args:
        data: same dict consumed by pts2render_cags.
        bg_color: background color list/tuple.
        level: one of:
          - 'all'           — base + all sub-Gaussians (equivalent to pts2render_cags)
          - 'base_only'     — only base (parent) Gaussians
          - 'drop_1' / 'drop_2' / 'drop_3'
                            — base + sub-Gaussians except those at sub-level k=j

    Returns:
        data with data['novel_view']['img_pred'] set, like pts2render_cags.

    Implementation: gates `sub_valid` per-level by leveraging the per-Gaussian
    level tag stored in data[view]['sub_level']  (B, N_sub) int tensor.
    The base (parent) is always level 0; sub-Gaussian k=j has level j.
    Each sub-level chunk has length H*W with k_sub chunks → level for index n
    is (n // (H*W)) + 1.
    """
    bs = data['lmain']['img'].shape[0]

    valid_overrides = {}
    for view in ('lmain', 'rmain'):
        N_sub = data[view]['sub_valid'].shape[1]
        H, W = data[view]['rot_maps'].shape[-2:]
        per_level = N_sub // (H * W)  # = k_sub

        sub_levels = torch.arange(N_sub, device=data[view]['sub_valid'].device) // (H * W) + 1
        sub_levels = sub_levels.unsqueeze(0).expand(bs, -1)  # (B, N_sub)

        original_sub_valid = data[view]['sub_valid']
        original_pts_valid = data[view]['pts_valid']

        if level == 'all':
            new_sub_valid = original_sub_valid
            new_pts_valid = original_pts_valid
        elif level == 'base_only':
            new_sub_valid = torch.zeros_like(original_sub_valid)
            new_pts_valid = original_pts_valid
        elif level.startswith('drop_'):
            j = int(level.split('_', 1)[1])
            assert 1 <= j <= per_level
            new_sub_valid = original_sub_valid & (sub_levels != j)
            new_pts_valid = original_pts_valid
        else:
            raise ValueError(f"unknown level={level!r}")

        valid_overrides[view] = (original_sub_valid, original_pts_valid)
        data[view]['sub_valid'] = new_sub_valid
        data[view]['pts_valid'] = new_pts_valid

    try:
        data = pts2render_cags(data, bg_color)
    finally:
        for view, (orig_sub, orig_pts) in valid_overrides.items():
            data[view]['sub_valid'] = orig_sub
            data[view]['pts_valid'] = orig_pts

    return data
```

```python
def l_disentangle(
    delta_images: Sequence[torch.Tensor],
    log_compress_k: float = 10.0,
) -> torch.Tensor:
    """Per-level sub-Gaussian band-specialization loss.

    For each j ∈ {1, 2, 3}, the contribution image ΔI_j of sub-Gaussian level j
    must concentrate entirely in detail band D_j; energy in LL or D_{i≠j} is
    penalized.

    L_disentangle = Σ_j  (‖LL_Δ_j‖₁ + Σ_{i≠j} ‖D_i_Δ_j‖₁)   in log-compressed space

    Args:
        delta_images: list of three (B, 3, H, W) tensors [ΔI_1, ΔI_2, ΔI_3].
        log_compress_k: log-compression strength.

    Returns:
        scalar tensor.
    """
    assert len(delta_images) == 3
    loss = delta_images[0].new_zeros(())
    for j_idx, delta in enumerate(delta_images, start=1):
        dwt, _ = _dwt3_padded(delta)
        loss = loss + log_compress(dwt["LL"], k=log_compress_k).abs().mean()
        for i in (1, 2, 3):
            if i == j_idx:
                continue
            for sub in dwt[f"D{i}"]:
                loss = loss + log_compress(sub, k=log_compress_k).abs().mean()
    return loss
```

```python
"""Cross-View Color Triangulation (CVCT) module for W-CVCT-GS.

Provides:
  - VisibilityGate:   per-pixel ω ∈ [0,1] (per-Gaussian visibility)
  - ResidualHead:     bounded Δrgb ∈ [-ε, ε]
  - CVCTModule:       assembles ColorEvidence + ω + Δrgb with identity-mode toggle
"""

from __future__ import annotations

from typing import Optional

import torch
import torch.nn as nn
import torch.nn.functional as F


class VisibilityGate(nn.Module):
    """Per-pixel visibility gate ω ∈ [0, 1].

    ω≈1 → trust this view's pixel; ω≈0 → trust the other view warped here;
    ω≈0.5 → both views agree (dual-visible region).
    """

    def __init__(self, in_channels: int, hidden: int = 32):
        super().__init__()
        self.net = nn.Sequential(
            nn.Conv2d(in_channels + 1, hidden, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(hidden, hidden, kernel_size=3, padding=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(hidden, 1, kernel_size=1),
        )

    def forward(self, fused_feat: torch.Tensor, confidence: torch.Tensor) -> torch.Tensor:
        """
        Args:
            fused_feat: (B, in_channels, H, W) — output of CrossViewFusion.
            confidence: (B, 1, H, W) — output of ConfidenceExtractor.
        Returns:
            omega: (B, 1, H, W) ∈ [0, 1].
        """
        x = torch.cat([fused_feat, confidence], dim=1)
        return torch.sigmoid(self.net(x))
```

To keep under 5K we use **two 3×3 + one 1×1** but we need to change the test to match real expectations. The spec actually allows ~3K — let me match the spec exactly: replace the second 3×3 with a 1×1 to compress.

```python
class ResidualHead(nn.Module):
    """Bounded RGB residual Δrgb ∈ [-ε, ε]."""

    def __init__(self, in_channels: int, hidden: int = 16, bound: float = 0.05):
        super().__init__()
        self.bound = float(bound)
        self.net = nn.Sequential(
            nn.Conv2d(in_channels, hidden, kernel_size=1),
            nn.ReLU(inplace=True),
            nn.Conv2d(hidden, 3, kernel_size=1),
        )

    def forward(self, shared_feat: torch.Tensor) -> torch.Tensor:
        """
        Args:
            shared_feat: (B, in_channels, H, W) — _shared_feat from FullResGaussianHead.
        Returns:
            delta_rgb: (B, 3, H, W) ∈ [-bound, +bound].
        """
        return torch.tanh(self.net(shared_feat)) * self.bound
```

```python
def warp_with_disparity(
    img: torch.Tensor,
    disp: torch.Tensor,
    padding_mode: str = "border",
) -> torch.Tensor:
    """Warp the *other* view's image to the current view using disparity.

    Mirrors lib.stereo_gs.cross_view_fusion.disparity_warp but operates on the
    raw RGB image. Assumes rectified stereo: x_other = x_self - disp.

    Args:
        img:  (B, 3, H, W) the other view's RGB.
        disp: (B, 1, H, W) disparity at the current view (in pixels).
        padding_mode: 'zeros' | 'border' | 'reflection'.

    Returns:
        (B, 3, H, W) warped image.
    """
    B, _, H, W = img.shape
    device = img.device
    grid_y = torch.arange(H, device=device, dtype=torch.float32).view(1, H, 1).expand(B, H, W)
    grid_x = torch.arange(W, device=device, dtype=torch.float32).view(1, 1, W).expand(B, H, W)
    src_x = grid_x - disp.squeeze(1)
    norm_x = 2.0 * src_x / (W - 1) - 1.0
    norm_y = 2.0 * grid_y / (H - 1) - 1.0
    grid = torch.stack([norm_x, norm_y], dim=-1)
    return F.grid_sample(img, grid, mode="bilinear", padding_mode=padding_mode, align_corners=True)


class CVCTModule(nn.Module):
    """Cross-View Color Triangulation: c_final = ω·c_self + (1-ω)·c_other_warp + Δrgb.

    `set_identity(True)` clamps ω=0.5 and Δrgb=0 (used in Phase 1 and Phase 2 of
    the training schedule, where CVCT must not interfere with FDSG learning).
    """

    def __init__(
        self,
        fused_channels: int,
        shared_channels: int,
        visibility_hidden: int = 32,
        residual_hidden: int = 16,
        residual_bound: float = 0.05,
    ):
        super().__init__()
        self.gate = VisibilityGate(in_channels=fused_channels, hidden=visibility_hidden)
        self.residual = ResidualHead(in_channels=shared_channels, hidden=residual_hidden, bound=residual_bound)
        self._identity_mode = False

    def set_identity(self, on: bool) -> None:
        self._identity_mode = bool(on)

    def is_identity(self) -> bool:
        return self._identity_mode

    def forward(
        self,
        fused_feat: torch.Tensor,
        shared_feat: torch.Tensor,
        confidence: torch.Tensor,
        c_self: torch.Tensor,
        c_other: torch.Tensor,
        disparity: torch.Tensor,
    ):
        """Compute per-Gaussian color from cross-view evidence.

        Args:
            fused_feat:  (B, C_fused, H, W).
            shared_feat: (B, C_shared, H, W) from FullResGaussianHead.
            confidence:  (B, 1, H, W).
            c_self:      (B, 3, H, W) RGB of THIS view in [0, 1].
            c_other:     (B, 3, H, W) RGB of OTHER view in [0, 1].
            disparity:   (B, 1, H, W) at THIS view (positive).

        Returns dict with:
            'c_final':         (B, 3, H, W) in [0, 1]
            'omega':           (B, 1, H, W)
            'delta_rgb':       (B, 3, H, W)
            'c_other_warped':  (B, 3, H, W) — used by L_cycle
        """
        c_other_warped = warp_with_disparity(c_other, disparity, padding_mode="border")

        if self._identity_mode:
            omega = torch.full_like(confidence, 0.5)
            delta_rgb = torch.zeros_like(c_self)
        else:
            omega = self.gate(fused_feat, confidence)
            delta_rgb = self.residual(shared_feat)

        c_final = omega * c_self + (1.0 - omega) * c_other_warped + delta_rgb
        c_final = c_final.clamp(0.0, 1.0)
        return {
            "c_final": c_final,
            "omega": omega,
            "delta_rgb": delta_rgb,
            "c_other_warped": c_other_warped,
        }
```

```python
"""CVCT losses for W-CVCT-GS.

  - l_cycle:        cross-view photometric consistency in dual-visible regions
  - l_omega_align:  visibility soft alignment to a target derived from
                    confidence and c-self / c-other agreement
"""

from __future__ import annotations

import torch
import torch.nn.functional as F


def _visibility_mask(omega: torch.Tensor) -> torch.Tensor:
    """4·ω·(1−ω) — peaks at ω=0.5, zero at endpoints."""
    return 4.0 * omega * (1.0 - omega)


def l_cycle(
    c_self: torch.Tensor,
    c_other_warped: torch.Tensor,
    omega: torch.Tensor,
) -> torch.Tensor:
    """Photometric consistency loss restricted to dual-visible regions.

    L_cycle = mean( visibility_mask · |c_self − c_other_warped| )

    Args:
        c_self, c_other_warped: (B, 3, H, W) in [0, 1].
        omega: (B, 1, H, W).

    Returns:
        scalar tensor.
    """
    mask = _visibility_mask(omega)
    diff = (c_self - c_other_warped).abs().mean(dim=1, keepdim=True)
    return (mask * diff).mean()


def build_omega_target(
    c_self: torch.Tensor,
    c_other_warped: torch.Tensor,
    confidence: torch.Tensor,
    agreement_sigma: float = 0.1,
) -> torch.Tensor:
    """Construct a soft target for ω from cross-view agreement and confidence.

    Heuristic:
      - High agreement (low |c_self - c_other_warped|) AND high confidence → target = 0.5
      - Low agreement OR low confidence → target shifts toward 1 (trust c_self)

    Args:
        c_self, c_other_warped: (B, 3, H, W) in [0, 1].
        confidence:             (B, 1, H, W) in [0, 1].
        agreement_sigma:        controls steepness of agreement exponent.

    Returns:
        omega_target: (B, 1, H, W) in [0, 1], detached.
    """
    diff = (c_self - c_other_warped).abs().mean(dim=1, keepdim=True)
    agreement = torch.exp(-(diff / agreement_sigma).pow(2))
    dual_visible = (agreement * confidence).clamp(0.0, 1.0)
    target = 0.5 + (1.0 - dual_visible) * 0.5
    return target.detach()


def l_omega_align(
    omega: torch.Tensor,
    omega_target: torch.Tensor,
    entropy_weight: float = 0.0,
) -> torch.Tensor:
    """L2 alignment of ω to a soft target, plus optional anti-collapse entropy.

    Args:
        omega:        (B, 1, H, W).
        omega_target: (B, 1, H, W) (typically detached).
        entropy_weight: optional coefficient on −H(ω) to prevent ω→0/1 collapse.

    Returns:
        scalar tensor.
    """
    align = F.mse_loss(omega, omega_target)
    if entropy_weight > 0:
        eps = 1e-6
        ent = -(omega.clamp(eps, 1 - eps) * omega.clamp(eps, 1 - eps).log()
                + (1 - omega).clamp(eps, 1 - eps) * (1 - omega).clamp(eps, 1 - eps).log())
        align = align - entropy_weight * ent.mean()
    return align
```

```python
"""Three-phase training schedule for W-CVCT-GS.

Phase 1 [0, phase1_end):           baseline only (RGB + Chamfer); CVCT identity.
Phase 2 [phase1_end, phase2_end):  +L_band, +L_active; CVCT still identity.
Phase 3 [phase2_end, ∞):           +L_disentangle (warmup), +L_cycle, +L_omega; CVCT active.
"""

from __future__ import annotations

from dataclasses import dataclass


@dataclass(frozen=True)
class PhaseState:
    phase: int                 # 1, 2, or 3
    cvct_identity_mode: bool
    lambda_band: float
    lambda_active: float
    lambda_disentangle: float
    lambda_cycle: float
    lambda_omega: float


class WCVCTSchedule:
    def __init__(
        self,
        phase1_end: int = 5000,
        phase2_end: int = 30000,
        lambda_band: float = 0.3,
        lambda_active: float = 0.05,
        lambda_disentangle: float = 0.5,
        lambda_disentangle_warmup: float = 0.1,
        lambda_disentangle_warmup_steps: int = 5000,
        lambda_cycle: float = 0.2,
        lambda_omega: float = 0.1,
    ):
        assert phase1_end < phase2_end
        self.phase1_end = phase1_end
        self.phase2_end = phase2_end
        self.lambda_band = lambda_band
        self.lambda_active = lambda_active
        self.lambda_disentangle = lambda_disentangle
        self.lambda_disentangle_warmup = lambda_disentangle_warmup
        self.lambda_disentangle_warmup_steps = lambda_disentangle_warmup_steps
        self.lambda_cycle = lambda_cycle
        self.lambda_omega = lambda_omega

    def state(self, step: int) -> PhaseState:
        if step < self.phase1_end:
            return PhaseState(
                phase=1, cvct_identity_mode=True,
                lambda_band=0.0, lambda_active=0.0,
                lambda_disentangle=0.0, lambda_cycle=0.0, lambda_omega=0.0,
            )
        if step < self.phase2_end:
            return PhaseState(
                phase=2, cvct_identity_mode=True,
                lambda_band=self.lambda_band, lambda_active=self.lambda_active,
                lambda_disentangle=0.0, lambda_cycle=0.0, lambda_omega=0.0,
            )
        # Phase 3
        steps_in_p3 = step - self.phase2_end
        if steps_in_p3 < self.lambda_disentangle_warmup_steps:
            ld = self.lambda_disentangle_warmup
        else:
            ld = self.lambda_disentangle
        return PhaseState(
            phase=3, cvct_identity_mode=False,
            lambda_band=self.lambda_band, lambda_active=self.lambda_active,
            lambda_disentangle=ld,
            lambda_cycle=self.lambda_cycle, lambda_omega=self.lambda_omega,
        )
```

```python
                    if n_sample < 100:
                        continue
                    idx_l = np.random.choice(l_xyz_v.shape[1], n_sample, replace=False)
                    idx_r = np.random.choice(r_xyz_v.shape[1], n_sample, replace=False)
                    cd_i, _ = chamfer_distance(l_xyz_v[:, idx_l], r_xyz_v[:, idx_r])
                    Lcd = Lcd + cd_i
                Lcd = Lcd / self.bs
                loss = loss + chamfer_weight * Lcd
                log['chamfer'] += chamfer_weight * Lcd.item()

            # ── W-CVCT-GS losses ──
            wcvct_logs = {}
            if phase_state is not None:
                loss, wcvct_logs = self._add_wcvct_losses(loss, data, phase_state)
            log.setdefault('band', 0.0); log['band'] += wcvct_logs.get('band', 0.0)
            log.setdefault('active', 0.0); log['active'] += wcvct_logs.get('active', 0.0)
            log.setdefault('dis', 0.0); log['dis'] += wcvct_logs.get('disentangle', 0.0)
            log.setdefault('cycle', 0.0); log['cycle'] += wcvct_logs.get('cycle', 0.0)
            log.setdefault('omega', 0.0); log['omega'] += wcvct_logs.get('omega', 0.0)

            log['l1'] += 0.8 * Ll1.item()
            log['ssim'] += 0.2 * Lssim.item()

            if metrics is None: metrics = {}
            metrics.update({'l1': Ll1.item(), 'ssim': Lssim.item(), 'chamfer': Lcd.item()})
            metrics.update({f'wcvct_{k}': v for k, v in wcvct_logs.items()})
            self.logger.push(metrics)

            self.scaler.scale(loss).backward()
            self.scaler.unscale_(self.optimizer)
            torch.nn.utils.clip_grad_norm_(
                [p for p in self.model.parameters() if p.requires_grad], 1.0)
            self.scaler.step(self.optimizer)
            if self.scheduler.last_epoch < self.cfg.num_steps + 99:
                self.scheduler.step()
            self.scaler.update()

            # logging blocks (unchanged) ...
            if self.total_steps and self.total_steps % self.cfg.record.loss_freq == 0:
                self.logger.writer.add_scalar(
                    'lr', self.optimizer.param_groups[0]['lr'], self.total_steps)
                if phase_state is not None:
                    self.logger.writer.add_scalar('wcvct/phase', phase_state.phase, self.total_steps)
            if self.total_steps and self.total_steps % LOG_PERIOD == 0:
                msg = "  ".join(f"{k}={v / LOG_PERIOD:.4f}" for k, v in log.items())
                logging.info(f"[step {self.total_steps}] {msg}")
                for k in log: log[k] = 0.0
            if self.total_steps and self.total_steps % self.cfg.record.eval_freq == 0:
                self.model.eval(); self.run_eval(); self.model.train(); self._freeze_bn()
            if self.total_steps > 0 and self.total_steps % self.cfg.record.save_iter == 0:
                self.save_ckpt(Path(f"{self.cfg.record.ckpt_path}/iter{self.total_steps}.pth"))
            self.total_steps += 1
```

```python
    def _add_wcvct_losses(self, loss, data, phase_state):
        """Compute and add W-CVCT-GS losses; returns (new_loss, log_dict)."""
        logs = {}
        if phase_state.lambda_band == 0.0 and phase_state.lambda_disentangle == 0.0 \
                and phase_state.lambda_active == 0.0 and phase_state.lambda_cycle == 0.0 \
                and phase_state.lambda_omega == 0.0:
            return loss, logs

        gt = data['novel_view']['img'].cuda()
        pred = data['novel_view']['img_pred']

        if phase_state.lambda_band > 0:
            Lb = l_band(
                pred, gt,
                ll_weight=self.wcvct_cfg.fdsg.ll_weight,
                band_weights=tuple(self.wcvct_cfg.fdsg.band_weights),
                log_compress_k=self.wcvct_cfg.fdsg.log_compress_k,
            )
            loss = loss + phase_state.lambda_band * Lb
            logs['band'] = phase_state.lambda_band * Lb.item()

        if phase_state.lambda_active > 0:
            extras = data.get('_stereo_gs_extras', {})
            wl = extras.get('split_weights_left'); wr = extras.get('split_weights_right')
            if wl is not None and wr is not None:
                gt_l = (data['lmain']['img'] * 0.5 + 0.5).clamp(0, 1)
                gt_r = (data['rmain']['img'] * 0.5 + 0.5).clamp(0, 1)
                La = 0.5 * (l_active(wl, gt_l) + l_active(wr, gt_r))
                loss = loss + phase_state.lambda_active * La
                logs['active'] = phase_state.lambda_active * La.item()

        if phase_state.lambda_disentangle > 0:
            deltas = self._compute_per_level_deltas(data)
            if deltas is not None:
                Ld = l_disentangle(
                    deltas, log_compress_k=self.wcvct_cfg.fdsg.log_compress_k)
                loss = loss + phase_state.lambda_disentangle * Ld
                logs['disentangle'] = phase_state.lambda_disentangle * Ld.item()

        if phase_state.lambda_cycle > 0 or phase_state.lambda_omega > 0:
            extras = data.get('_stereo_gs_extras', {})
            for view_key, prefix in (('cvct_left', 'l'), ('cvct_right', 'r')):
                cvct_out = extras.get(view_key)
                if cvct_out is None: continue
                if phase_state.lambda_cycle > 0:
                    Lc = l_cycle(
                        c_self=(data['lmain' if prefix == 'l' else 'rmain']['img'] * 0.5 + 0.5).clamp(0, 1),
                        c_other_warped=cvct_out['c_other_warped'],
                        omega=cvct_out['omega'],
                    )
                    loss = loss + phase_state.lambda_cycle * Lc
                    logs[f'cycle_{prefix}'] = phase_state.lambda_cycle * Lc.item()
                if phase_state.lambda_omega > 0:
                    conf_key = 'confidence_left' if prefix == 'l' else 'confidence_right'
                    conf = extras.get(conf_key)
                    if conf is not None:
                        H, W = cvct_out['omega'].shape[-2:]
                        if conf.shape[-1] != W:
                            conf = F.interpolate(conf, size=(H, W), mode='bilinear', align_corners=False)
                        c_self = (data['lmain' if prefix == 'l' else 'rmain']['img'] * 0.5 + 0.5).clamp(0, 1)
                        target = build_omega_target(c_self, cvct_out['c_other_warped'], conf)
                        Lo = l_omega_align(
                            cvct_out['omega'], target,
                            entropy_weight=self.wcvct_cfg.cvct.lambda_omega_entropy,
                        )
                        loss = loss + phase_state.lambda_omega * Lo
                        logs[f'omega_{prefix}'] = phase_state.lambda_omega * Lo.item()
        # aggregate split logs
        for agg_key in ('cycle', 'omega'):
            keys = [k for k in logs if k.startswith(agg_key + '_')]
            if keys:
                logs[agg_key] = sum(logs[k] for k in keys) / len(keys)
        return loss, logs

    def _compute_per_level_deltas(self, data):
        """Render four times to extract ΔI_j contribution images for L_disentangle.

        Phase A naive: 4 renders per step. Returns list [ΔI_1, ΔI_2, ΔI_3] or None.
        """
        bg = self.cfg.dataset.bg_color
        # We must NOT mutate data['novel_view']['img_pred'] permanently — save & restore.
        original_pred = data['novel_view']['img_pred']
        try:
            data = pts2render_cags_per_level(data, bg, level='all')
            I_full = data['novel_view']['img_pred']
            deltas = []
            for j in (1, 2, 3):
                data = pts2render_cags_per_level(data, bg, level=f'drop_{j}')
                I_drop = data['novel_view']['img_pred']
                deltas.append(I_full - I_drop)
            data['novel_view']['img_pred'] = original_pred  # restore
            return deltas
        except Exception as e:
            logging.warning(f"[W-CVCT-GS] L_disentangle render failed: {e}; skipping")
            data['novel_view']['img_pred'] = original_pred
            return None
```
